export async function generateMetadata({ params }) {
  const dict = await getDictionary(params.lang)
  const ourBrandLabel = { mn: 'Манай брэнд', ru: 'Наш бренд', en: 'Our brand' }
  return {
    title: `${caseStudy.client} — ${ourBrandLabel[params.lang] ?? ourBrandLabel.mn}`,
    description: dict.work?.['estel']?.description ?? caseStudy.description,
  }
}
